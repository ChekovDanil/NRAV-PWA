    );
  };

  const readViewportState = () => {
    const layoutHeight = Math.max(window.innerHeight, root.clientHeight);
    const viewport = window.visualViewport;
    if (!viewport || viewport.height <= 0) return { height: Math.ceil(layoutHeight), keyboardRaised: false };

    // In an installed iOS web app with viewport-fit=cover and a translucent
    // status bar, VisualViewport can start below y=0. Its height alone then
    // shortens the application shell and exposes a strip at the bottom. The
    // visible bottom edge is height + offsetTop. Only an actually open keyboard
    // is allowed to make the shell shorter than the layout viewport.
    const visualBottom = viewport.height + Math.max(0, viewport.offsetTop);
    const keyboardRaised = isEditableElementActive()
      && layoutHeight - viewport.height > 120;
    return {
      // Safari may pan the visual viewport while focusing an input. The keyboard
      // edge is then at offsetTop + height in layout coordinates, so anchoring
      // the shell to height alone lifts the composer far above the keyboard.
      height: Math.ceil(keyboardRaised ? visualBottom : Math.max(layoutHeight, visualBottom)),
      keyboardRaised,
    };
  };

  const commit = () => {
    animationFrame = 0;
    const state = readViewportState();
    root.style.setProperty(property, state.height + "px");
    if (state.keyboardRaised) root.style.setProperty(safeAreaProperty, "0px");
    else root.style.removeProperty(safeAreaProperty);
  };

  const schedule = () => {
    if (animationFrame) cancelAnimationFrame(animationFrame);
    animationFrame = requestAnimationFrame(commit);
  };

  const scheduleSettledRead = () => {
    schedule();
    if (settleTimer) clearTimeout(settleTimer);
    settleTimer = window.setTimeout(schedule, 240);
  };

  const handleVisibilityChange = () => {
    if (document.visibilityState === "visible") scheduleSettledRead();
  };

  window.addEventListener("orientationchange", scheduleSettledRead);
  window.addEventListener("pageshow", scheduleSettledRead);
  window.visualViewport?.addEventListener("resize", schedule);
  window.visualViewport?.addEventListener("scroll", schedule);
  document.addEventListener("visibilitychange", handleVisibilityChange);
